export const describeTransition = (transition) =>
	transition.type === "spring"
		? `spring · stiffness ${transition.stiffness} · damping ${transition.damping}`
		: `${Math.round(transition.duration * 1000)} ms · cubic-bezier(${transition.ease.join(", ")})`;

export const GestureDemos = () => {
	const { prefersReducedMotion, resolve } = useMotionPreference();
	const [confirmKey, setConfirmKey] = React.useState(0);
	return (
		<div style={{ display: "flex", gap: "1.5rem", flexWrap: "wrap", margin: "1.5rem 0" }}>
			<DemoSurface {...(prefersReducedMotion ? {} : liftPattern)}>Lift: hover me</DemoSurface>
			<DemoSurface whileTap={pressPattern.whileTap} transition={resolve(pressPattern.transition)}>
				Press: click me
			</DemoSurface>
			<DemoSurface
				key={confirmKey}
				animate={prefersReducedMotion ? undefined : confirmPattern.animate}
				transition={confirmPattern.transition}
				onClick={() => setConfirmKey((key) => key + 1)}
			>
				Confirm: click me
			</DemoSurface>
		</div>
	);
};

export const RevealDemo = () => {
	const { resolve } = useMotionPreference();
	const [shown, setShown] = React.useState(true);
	return (
		<div style={{ margin: "1.5rem 0" }}>
			<button
				type='button'
				onClick={() => setShown((value) => !value)}

						<motion.div

							animate='animate'
							exit='exit'

								width: 240,

								background: "var(--secondary)",

						</motion.div>
					)}

		</div>
	);

# Motion

Motion explains a change in state, keeps spatial continuity and makes the interface feel responsive. It is never decoration. The personality is **calm, refined, natural and responsive**: never bouncy, cartoonish, mechanical or distracting. When unsure, choose the more subtle option.

The rules are in `MOTION.md`. The values on this page are read from `src/lib/motion/`, so they always match the code.

<Source
	language='tsx'
	code={`import { motion } from "motion/react";
import { pressPattern, useMotionPreference } from "@jcaunedo/intercept-ucl";

const { resolve } = useMotionPreference();

<motion.button whileTap={pressPattern.whileTap} transition={resolve(pressPattern.transition)} />`}
/>

Never write a raw duration, easing, distance or scale in a component. Use these exports, and add any new value to `tokens.ts` with a comment saying what it's for.

## Durations

Nothing goes past `deliberate`. Exits are shorter than entrances.

## Transitions

| Transition           | Value                                         | Use                                         |
| -------------------- | --------------------------------------------- | ------------------------------------------- |
| `standardTransition` | {describeTransition(standardTransition)}      | Common UI changes                           |
| `enterTransition`    | {describeTransition(enterTransition)}         | Elements arriving: quick start, gentle finish |
| `exitTransition`     | {describeTransition(exitTransition)}          | Elements leaving: short and decisive        |
| `subtleSpring`       | {describeTransition(subtleSpring)}            | Small, contained motion (press, indicators) |
| `responsiveSpring`   | {describeTransition(responsiveSpring)}        | Direct manipulation that should feel snappy |
| `layoutSpring`       | {describeTransition(layoutSpring)}            | Layout and shared-element movement          |

Springs are heavily damped, so they never visibly bounce.

## Patterns

Use a named pattern before inventing new motion.

| Pattern | Use for                                  | Implementation                                          |
| ------- | ---------------------------------------- | ------------------------------------------------------- |
| Lift    | Hoverable cards and surfaces             | `liftPattern`, `liftWithMediaPattern`                   |
| Press   | Buttons, cards, direct interactions      | `pressPattern`                                          |
| Reveal  | Content appearing in context             | `revealVariants`                                        |
| Expand  | A region opening or closing in place     | `expandVariants` + `useExpandClip()`                    |
| Connect | Shared-element transitions               | `layoutId` + `layoutSpring`                             |
| Flow    | Small groups entering together           | `staggerContainerVariants` + children `revealVariants`  |
| Glide   | Directional navigation                   | `slideVariants`, `glidePattern`                         |
| Dismiss | Content removed or closed                | `dismissVariants`; table and list rows: `rowRemoveVariants` |
| Confirm | Lightweight success feedback             | `confirmPattern`                                        |
| Pending | Work in progress on an element's behalf  | `pendingIconVariants`, `pendingSurfaceVariants`         |

Also available: `fadeVariants`, `overlayVariants` for scrims, and `panelVariants` for drawers.

## Accessibility

- Always respect reduced motion with `useMotionPreference()` and its `resolve()` helper.
- Under reduced motion, remove nonessential movement and prefer opacity. State-change feedback stays, but becomes near-instant.
- Motion is never the only signal of a change. Keyboard behavior, focus order and visible focus must survive every animation.
